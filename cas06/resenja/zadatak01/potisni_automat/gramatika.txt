---------------------------------------------------------------------------------------------
Pocetna gramatika: 
---------------------------------------------------------------------------------------------

NIZ_NAREDBI -> NIZ_NAREDBI NAREDBA ;
            | NAREDBA ;

NAREDBA -> id = E
            | print ( E )

E -> E + T
      | E - T
      | T

T -> T * F
      | T / F
      | F

F -> ( E )                                   
      | broj                                 
      | id                                   
      | - F                                   

---------------------------------------------------------------------------------------------

      | eps                                  {eoi}
